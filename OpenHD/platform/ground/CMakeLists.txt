if(MSVC)
    message(FATAL_ERROR "The ground backend currently requires MinGW-w64 (GCC), due to wifibroadcast packed wire structures")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WB_DEVOURER_ONLY ON CACHE BOOL "" FORCE)
set(WB_USE_SPDLOG_EXTERNALLY OFF CACHE BOOL "" FORCE)
set(ground_root "${CMAKE_CURRENT_LIST_DIR}/../..")
find_package(PkgConfig REQUIRED)
pkg_check_modules(libusb REQUIRED IMPORTED_TARGET GLOBAL libusb-1.0)
add_subdirectory(${ground_root}/ohd_interface/lib/devourer devourer EXCLUDE_FROM_ALL)
# Devourer uses both <libusb.h> and <libusb-1.0/libusb.h>.
get_target_property(usb_includes PkgConfig::libusb INTERFACE_INCLUDE_DIRECTORIES)
foreach(usb_include IN LISTS usb_includes)
    get_filename_component(usb_include_parent "${usb_include}" DIRECTORY)
    target_include_directories(devourer PUBLIC "${usb_include_parent}")
endforeach()
# Use OpenHD's pinned wifibroadcast fork for the wire protocol. Devourer is
# its USB radio transport, just as in the Linux OpenHD integration.
include(${ground_root}/ohd_interface/lib/wifibroadcast/wifibroadcast/WBLib.cmake)
target_include_directories(wifibroadcast PUBLIC
    ${ground_root}/ohd_interface/lib/wifibroadcast/wifibroadcast/src)

add_library(openhd_ground_config STATIC ground_config.cpp
    ${ground_root}/ohd_interface/src/devourer_usb_device_probe.cpp)
target_include_directories(openhd_ground_config PUBLIC ${CMAKE_CURRENT_LIST_DIR}
    ${ground_root}/ohd_interface/inc ${ground_root}/ohd_common/inc ${ground_root}/ohd_common/lib/json/include)
target_link_libraries(openhd_ground_config PUBLIC devourer)

add_library(openhd_ground_udp STATIC
    ${ground_root}/ohd_common/src/openhd_udp.cpp console_log.cpp ground_settings_ui.cpp
    ${ground_root}/ohd_common/src/openhd_ncurses_ui.cpp)
target_include_directories(openhd_ground_udp PUBLIC
    ${ground_root}/ohd_common/inc ${ground_root}/ohd_common/lib/spdlog/include)
find_package(Threads REQUIRED)
target_link_libraries(openhd_ground_udp PUBLIC Threads::Threads openhd_ground_config)
target_compile_definitions(openhd_ground_udp PRIVATE OPENHD_HAVE_CURSES OPENHD_PORTABLE_GROUND)
if(WIN32)
    target_link_libraries(openhd_ground_config PUBLIC ws2_32)
    find_path(PDCURSES_INCLUDE_DIR curses.h PATH_SUFFIXES pdcurses REQUIRED)
    find_library(PDCURSES_LIBRARY NAMES pdcurses_wincon REQUIRED)
    target_include_directories(openhd_ground_udp PUBLIC ${PDCURSES_INCLUDE_DIR})
    target_compile_definitions(openhd_ground_udp PUBLIC PDC_WIDE PDC_FORCE_UTF8)
    target_link_libraries(openhd_ground_udp PUBLIC ${PDCURSES_LIBRARY} iphlpapi user32 winmm)
    target_compile_definitions(openhd_ground_udp PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_definitions(wifibroadcast PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(openhd_ground_udp PUBLIC ws2_32)
else()
    find_package(Curses REQUIRED)
    target_include_directories(openhd_ground_udp PUBLIC ${CURSES_INCLUDE_DIRS})
    target_link_libraries(openhd_ground_udp PUBLIC ${CURSES_LIBRARIES})
endif()
add_executable(openhd_ground main.cpp)
target_link_libraries(openhd_ground PRIVATE openhd_ground_udp wifibroadcast)
if(OPENHD_ENABLE_GUI)
    target_link_libraries(openhd_ground PRIVATE openhd_gui)
endif()
install(TARGETS openhd_ground RUNTIME DESTINATION bin)

add_executable(test_ground_udp test_udp.cpp)
target_link_libraries(test_ground_udp PRIVATE openhd_ground_udp)
add_test(NAME ground_udp COMMAND test_ground_udp)
set_tests_properties(ground_udp PROPERTIES TIMEOUT 15)
add_test(NAME ground_help COMMAND openhd_ground --help)
add_executable(test_ground_pipeline test_pipeline.cpp)
target_link_libraries(test_ground_pipeline PRIVATE wifibroadcast)
add_test(NAME ground_pipeline COMMAND test_ground_pipeline)
set_tests_properties(ground_pipeline PROPERTIES TIMEOUT 15)
add_executable(test_ground_dashboard test_dashboard.cpp)
target_link_libraries(test_ground_dashboard PRIVATE openhd_ground_udp)
if(WIN32)
    add_test(NAME ground_dashboard COMMAND test_ground_dashboard ${CMAKE_CURRENT_BINARY_DIR}/dashboard.txt)
    set_tests_properties(ground_dashboard PROPERTIES TIMEOUT 25)
    add_executable(test_ground_app test_app.cpp)
    target_link_libraries(test_ground_app PRIVATE openhd_ground_config)
    add_test(NAME ground_app COMMAND test_ground_app $<TARGET_FILE:openhd_ground> ${CMAKE_CURRENT_BINARY_DIR}/app-test)
    set_tests_properties(ground_app PROPERTIES TIMEOUT 25)
endif()
